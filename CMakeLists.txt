cmake_minimum_required(VERSION 3.20)
# 需要 C：third_party/sqlite 是 .c 文件（只声明 CXX 的话 CMake 会直接不收它）
# 需要 RC：res/resources.rc 把 exe 图标、关于页 logo、图标字体编进 exe
project(meow-desk-core LANGUAGES C CXX RC)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 不依赖 core-ui：它的主窗口 swap chain 不带 alpha，窗口无法半透明
# （实测 SetLayeredWindowAttributes / DwmEnableBlurBehindWindow /
#   SetWindowCompositionAttribute 全部无效），且它的 D3D11+D2D 管线
#   固定占用约 49MB 私有内存。
# 桌面层、盒子、拖拽层全部改用纯 Win32 UpdateLayeredWindow + GDI 自绘。

# ---------------------------------------------------------------------------
# 版本资源（右键 → 属性 → 详细信息读的就是它）
#
# exe 里没有 RT_VERSION / VS_VERSION_INFO 时，Windows 11 会显示
# 「此文件没有可用的版本信息」，文件版本 / 产品名称 / 版权全是空的。
# 光设 CMAKE_PROJECT_VERSION 没用 —— 那只影响 CMake 内部变量，不会写进 PE。
#
# 版本号唯一来源是 version.txt（打包产物名用的也是它，make-setup --version
# 会写回），所以这里在 configure 阶段按它生成 rc，别在 rc 里手写死版本，
# 否则必然出现「包叫 MeowDesk_v0.2.0、属性里还是 0.1.0」。
# ---------------------------------------------------------------------------
set(MEOW_VERSION_FILE "${CMAKE_CURRENT_SOURCE_DIR}/version.txt")
# 光改 version.txt 时 cmake --build 不会重新 configure，加这条才会
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${MEOW_VERSION_FILE}")

if(EXISTS "${MEOW_VERSION_FILE}")
    file(READ "${MEOW_VERSION_FILE}" MEOW_VERSION)
    string(STRIP "${MEOW_VERSION}" MEOW_VERSION)
endif()
if(NOT MEOW_VERSION MATCHES "^[0-9]+(\\.[0-9]+)*$")
    message(WARNING "version.txt 内容不合法（要 0.1.0 这种形式），本次按 0.0.0 处理")
    set(MEOW_VERSION "0.0.0")
endif()

# FILEVERSION / PRODUCTVERSION 要 4 段 16 位数字：0.1.0 → 0,1,0,0
string(REPLACE "." ";" _meow_ver_parts "${MEOW_VERSION}")
list(LENGTH _meow_ver_parts _meow_ver_n)
while(_meow_ver_n LESS 4)
    list(APPEND _meow_ver_parts 0)
    math(EXPR _meow_ver_n "${_meow_ver_n} + 1")
endwhile()
while(_meow_ver_n GREATER 4)
    list(REMOVE_AT _meow_ver_parts -1)
    math(EXPR _meow_ver_n "${_meow_ver_n} - 1")
endwhile()
string(JOIN "," MEOW_VERSION_QUAD ${_meow_ver_parts})

# 生成一个 VERSIONINFO rc。两个 exe 只有 FileDescription / OriginalFilename
# 不同，所以共用一个函数。
#
# 坑：rc.exe 默认按系统 ANSI（本机是 GBK）读 .rc，UTF-8 的中文会变成乱码
# 存进资源里 —— 而中文的「值」写不成 ASCII 转义，所以必须靠
# #pragma code_page(65001) 让 rc 按 UTF-8 解析（实测有效，无需额外编译器 flag）。
function(meow_write_version_rc rc_path original_name file_desc)
    file(WRITE "${rc_path}" "\
#pragma code_page(65001)
// generated by CMake from version.txt -- do not edit
1 VERSIONINFO
FILEVERSION ${MEOW_VERSION_QUAD}
PRODUCTVERSION ${MEOW_VERSION_QUAD}
FILEFLAGSMASK 0x3fL
FILEFLAGS 0x0L
FILEOS 0x40004L
FILETYPE 0x1L
FILESUBTYPE 0x0L
BEGIN
    BLOCK \"StringFileInfo\"
    BEGIN
        BLOCK \"080404b0\"
        BEGIN
            VALUE \"CompanyName\", \"MeowDesk\"
            VALUE \"FileDescription\", \"${file_desc}\"
            VALUE \"FileVersion\", \"${MEOW_VERSION}\"
            VALUE \"InternalName\", \"MeowDesk\"
            VALUE \"LegalCopyright\", \"Copyright (C) 2026 MeowDesk\"
            VALUE \"OriginalFilename\", \"${original_name}\"
            VALUE \"ProductName\", \"MeowDesk\"
            VALUE \"ProductVersion\", \"${MEOW_VERSION}\"
        END
    END
    BLOCK \"VarFileInfo\"
    BEGIN
        VALUE \"Translation\", 0x804, 1200
    END
END
")
    set_source_files_properties("${rc_path}" PROPERTIES GENERATED TRUE)
    # 版本号变了要重编 rc（configure 会重写文件，.d 里也看不到 version.txt）
    set_source_files_properties("${rc_path}" PROPERTIES OBJECT_DEPENDS
        "${MEOW_VERSION_FILE}")
endfunction()

set(APP_VERSION_RC   "${CMAKE_CURRENT_BINARY_DIR}/version_app.rc")
set(SETUP_VERSION_RC "${CMAKE_CURRENT_BINARY_DIR}/version_setup.rc")
meow_write_version_rc("${APP_VERSION_RC}" "MeowDesk.exe"
    "MeowDesk 桌面分区整理工具")
# 安装器的 OriginalFilename 跟分发包里的实际文件名对齐（package.py 那套命名）
meow_write_version_rc("${SETUP_VERSION_RC}" "MeowDesk_v${MEOW_VERSION}_setup.exe"
    "MeowDesk 安装程序")

add_executable(meow-desk WIN32
    src/main.cpp
    src/desktop_icons.cpp
    src/shell_icons.cpp
    src/thumb_loader.cpp
    src/icon_render.cpp
    src/desktop_surface.cpp
    src/drag_layer.cpp
    src/fence_window.cpp
    src/backdrop.cpp
    src/config.cpp
    src/storage.cpp
    src/zorder.cpp
    src/tray.cpp
    src/screen_util.cpp
    src/settings_window.cpp
    src/search_window.cpp
    src/popup_menu.cpp
    src/confirm_dialog.cpp
    src/pinyin.cpp
    src/layout_store.cpp
    src/file_drop.cpp
    src/rename_box.cpp
    src/rename_dialog.cpp
    src/shell_drag.cpp
    src/app_icon.cpp
    src/icon_font.cpp
    # exe 图标 / 关于页 logo / 设置页图标字体（见 res/app_res.h）
    res/resources.rc
    # 右键属性里的版本信息（configure 阶段按 version.txt 生成）
    "${APP_VERSION_RC}"
    # 本地存储用 SQLite 3 的 amalgamation（单文件，无外部依赖）。
    # 比 *.ini 整份重写可靠：事务保证原子性，加字段走 user_version 迁移。
    # 内存代价实测为零 —— 代码页是 file-backed，不占 private。
    third_party/sqlite/sqlite3.c
)

target_include_directories(meow-desk PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}/src"
    "${CMAKE_CURRENT_SOURCE_DIR}/res"
    "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite"
)

# sqlite3.c 是一份 9MB 的第三方 amalgamation，别让 /W3 把时间浪费在它身上
if(MSVC)
    set_source_files_properties(third_party/sqlite/sqlite3.c PROPERTIES
        COMPILE_OPTIONS "/w")
endif()

# rc.exe 引用的那些图片/字体**不会**被自动跟踪：cmcldeps 生成的 .d 只记 rc 里
# #include 的头文件，不记 ICON / RCDATA 指向的文件。结果是「换了 ico 或 png，
# ninja 认为资源没变，exe 里还是旧图标」—— 而且完全没有任何报错，只有肉眼看出来。
# 这里手动把它们声明成依赖。
set_source_files_properties(res/resources.rc PROPERTIES OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/res/meow-desk.ico;${CMAKE_CURRENT_SOURCE_DIR}/res/logo-128.png;${CMAKE_CURRENT_SOURCE_DIR}/src/iconfont/iconfont.ttf"
)
set_source_files_properties(res/setup_res.rc PROPERTIES OBJECT_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/res/meow-desk.ico"
)

# /MD：与 GDI+ / UCRT 导入符号对齐
set_property(TARGET meow-desk PROPERTY
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

target_link_libraries(meow-desk PRIVATE
    gdiplus user32 gdi32 shell32 ole32 shlwapi dwmapi comctl32 msimg32 uxtheme imm32
)

target_compile_definitions(meow-desk PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)

# SQLite：单线程使用（只有 UI 线程碰它），砍掉用不到的特性；
# DQS=0 关掉「双引号当字符串」的兼容行为；MEMSTATUS=0 省掉每次 malloc 的统计。
target_compile_definitions(meow-desk PRIVATE
    SQLITE_THREADSAFE=0
    SQLITE_OMIT_LOAD_EXTENSION
    SQLITE_DQS=0
    SQLITE_DEFAULT_MEMSTATUS=0
    SQLITE_OMIT_DEPRECATED
    SQLITE_OMIT_SHARED_CACHE
)

if(MSVC)
    target_compile_options(meow-desk PRIVATE /W3 /utf-8)
endif()

# ---------------------------------------------------------------------------
# 安装程序：把主程序复制到用户选定位置，并决定数据放 %APPDATA% 还是同目录（便携）。
#
# 主程序**编进安装器自己的资源里**（IDR_PAYLOAD）。以前是「读 setup.exe 同目录的
# MeowDesk.exe」，但下面两种情况会让它找不到文件：
#   · 在资源管理器里直接双击 zip 中的 setup —— Windows 只把被双击的那一个文件
#     解到 %TEMP% 再运行，同目录没有主程序；
#   · 把 setup 单独拷到别处。
# 现在一个 setup.exe 就能装完；分发包里那份 MeowDesk.exe 退化成
# 「免安装直接跑」的可选件（想放在同目录给它读也行，见 setup.cpp 的回退分支）。
# ---------------------------------------------------------------------------

# payload.rc：内容只有一行 RCDATA，指向构建产物 meow-desk.exe，所以只能在
# configure 阶段按构建目录拼出来（源树里没法写死路径）。
# 用正斜杠：rc 的字符串会处理反斜杠转义，写 C:\Users 会被当成 \U。
set(PAYLOAD_RC "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
file(WRITE "${PAYLOAD_RC}"
    "// generated by CMake -- 主程序打包进安装器\n"
    "#include \"app_res.h\"\n"
    "IDR_PAYLOAD RCDATA \"${CMAKE_CURRENT_BINARY_DIR}/meow-desk.exe\"\n")
set_source_files_properties("${PAYLOAD_RC}" PROPERTIES GENERATED TRUE)

add_executable(meow-setup WIN32 setup/setup.cpp res/setup_res.rc
    "${SETUP_VERSION_RC}" "${PAYLOAD_RC}")

# 顺序：主程序链接完才有 meow-desk.exe 可读（否则 rc 报 cannot open file）。
add_dependencies(meow-setup meow-desk)

# 同上文的 OBJECT_DEPENDS 问题：rc 不跟踪 RCDATA 指向的文件，改了主程序
# 而 ninja 认为 payload.rc 没变 → 安装器里还是旧的主程序。必须手动声明依赖，
# 否则「装了新版但跑起来还是老界面」，而且没有任何报错。
set_source_files_properties("${PAYLOAD_RC}" PROPERTIES OBJECT_DEPENDS
    "${CMAKE_CURRENT_BINARY_DIR}/meow-desk.exe"
)
# include "app_res.h"
target_include_directories(meow-setup PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/res")

set_property(TARGET meow-setup PROPERTY
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
target_link_libraries(meow-setup PRIVATE user32 gdi32 shell32 ole32 shlwapi comctl32 dwmapi)
target_compile_definitions(meow-setup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)

if(MSVC)
    target_compile_options(meow-setup PRIVATE /W3 /utf-8)
endif()
